<div style="max-width:100%;overflow:hidden">
  <iframe id="gpPrice"
          src="https://ВАШ-ЛОГИН.github.io/ВАШ-РЕПОЗИТОРИЙ/price.html"
          title="Прайс «Георгин»"
          scrolling="no"
          style="display:block;width:100%;height:3000px;border:0;overflow:hidden"></iframe>
</div>
<script>
(function () {
  var f = document.getElementById('gpPrice');
  var SRC = 'https://ВАШ-ЛОГИН.github.io';  // домен iframe, без пути
  var HEADER = 0;                            // высота фиксированной шапки Tilda, px
  window.addEventListener('message', function (e) {
    if (e.origin !== SRC) return;
    var d = e.data;
    if (!d || !d.gpPrice) return;
    if (d.gpPrice === 'height' && d.height) {
      f.style.height = d.height + 'px';
    }
    if (d.gpPrice === 'scroll') {
      var y = f.getBoundingClientRect().top + window.pageYOffset + (d.top || 0) - HEADER;
      window.scrollTo(0, y);
    }
  });
})();
</script>
